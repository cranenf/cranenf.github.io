变量名解释: $a_{i,u}$ 存是否被咬坏，是为 $1$ ，不是为 $0$ 。 $ma_{i,u}$ 存当前点向下延伸的最大三角形的边长

题意不做解释，没有难点。

一开始看懂题目后的想法是要从上往下搜，可以搜出每个点往下延伸的最大的三角形是多大。随即意识到既然可以从上往下搜索，就可以从下往上递推。注意，倒过来的三角形也要倒过来处理，包括下面的求解。

判断方法很简单，假如当前在 $a_
{i,u}$ ,则只需要判断 $a_{i,u}$ 、 $a_{i + 1,u}$ 、 $a_{i + 1,u + 1}$ 、 $a_{i + 1,u + 2}$ 中是否有 $1$ 。若有，则 $ma_{i,u} = 0$ ,否则就在 $ma_{i + 1,u}$ 、 $ma_{i + 1，u+ 2}$ 中选最小的，原因如下：

这样就可以处理出每一个点的 $ma$ 值了。然后考虑如何取出两个使得和最大且不相交。一开始我觉得对于每个点，可以到它所延伸的三角形的下面去找一个最大的和它相加。又想想其实不止下面，这一个位置也不一定要取到最大。但这种分割两部分的思想可以保留，上下各找一个不超过分界线的最大三角形相加。

因此，可以枚举横向的分界线，数据量较小，可以暴力去求上下部分的最大。但很明显，两个三角形不一定是上下排布，左右也可以，斜着也行，但这条分界线是一直在的。而我们上面求ma都是上下的，所以可以把整个数组数组旋转一下，就可以处理这种情况了。至于旋转的方法，找规律就行了 [这篇题解](https://www.luogu.com.cn/article/m9e7t4pv)

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1001;
int n,m,ans;
int a[N][N],dp[N][N],ma[N][N],te[N][N];
void solve()
{
	memset(ma,0,sizeof ma);
	for(int i = 1;i <= n;i++)
		for(int u = 1;u < i * 2;u++)//预处理，不过这样写会有点繁琐 
			if(!a[i][u])
				ma[i][u] = 1;
	for(int i = n - 1;i >= 1;i--)
		for(int u = 1;u < i * 2;u += 2)
			if(!(a[i][u] || a[i + 1][u] || a[i + 1][u + 1] || a[i + 1][u + 2]))//求解ma，每个位置向下最大可以延伸的三角形边长 
				ma[i][u] = min(ma[i + 1][u],ma[i + 1][u + 2]) + 1;
	for(int i = 2;i <= n;i++)
		for(int u = 2;u < i * 2;u += 2)
			if(!(a[i][u] || a[i - 1][u] || a[i - 1][u - 1] || a[i - 1][u - 2]))
				ma[i][u] = min(ma[i - 1][u],ma[i - 1][u - 2]) + 1;
	int res = 0,up,dn;
	for(int i = 1;i <= n;i++)//枚举分界线，找最优解 
	{
		up = dn = 0;
		for(int u = 1;u <= i;u++)//暴力上半部分找最大 
		{
			for(int o = 1;o < u * 2;o += 2)//枚举正向的三角形，即向下延伸的 
				up = max(up,min(i - u + 1,ma[u][o]));//因为可能向下延伸到分割线下，所以加个min，如果超了就取最大可以取的（指不超过分界线 
			for(int o = 2;o < u * 2;o += 2)//枚举倒向的三角形，即向上延伸的 
				up = max(up,ma[u][o]);
		}
		for(int u = i + 1;u <= n;u++)//暴力下半部分 
		{
			for(int o = 1;o < u * 2;o += 2)
				dn = max(dn,ma[u][o]);
			for(int o = 2;o < u * 2;o += 2)
				dn = max(dn,min(u - i,ma[u][o]));//同上，可能向上延伸到分割线以上 
		}
		if(up && dn)//！！！！注意！！！有可能只取得到一个点，此时不能更新！！！ 
			res = max(res,up * up + dn * dn);
	}
	ans = max(ans,res);
}
void revolve()//旋转，虽然我的写法不同，但思路是一样的 
{
	int x = n,y = 1; 
	for(int u = 1;u < 2 * n;u++)
	{
		for(int i = u / 2 + 1;i <= n;i++)
		{
			te[x][y] = a[i][u];
			y += 2;
			if(y == 2 * x + 1)
				y = 2;
			else if(y == 2 * x)
			{
				x--;
				y = 1;
			}
		}
	}
	memcpy(a,te,sizeof te);
}
int main()
{
	int t,t1;
	cin >> n >> m;
	for(int i = 1;i <= m;i++)
	{
		cin >> t >> t1;
		a[t][t1] = 1;
	}
	solve();//转两次求三次，每个方向求一次 
	revolve();
	solve();
	revolve();
	solve();
	cout << ans;
	return 0;
}
```
