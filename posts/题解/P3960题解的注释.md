#include<iostream>

template<class T>inline bool chkmax(T&x,T y) { return x < y ? x = y,1 : 0; }

template<typename T>inline void read(T&x)//快读 
{
    x = 0;char c;int sign = 1;

    do { x = x * 10 + c - '0'; c = getchar();  }while(c <= '9' && c >= '0');
    x *= sign;
}

{

const int N = 3e5 + 500,M = 1e7;
int n,m,now,o[N],q,root[N],SZ,sz[M],ls[M],rs[M];ll val[M];//now是当前操作的是哪棵树    root是每颗树的根节点编号   sz是每个节点子树大小   val是人的编号   o是当前树已经多开了几个点 
inline int get_sz(int l,int r)
{
    if(now == n + 1)
    {
        if(r <= n) return r - l + 1;
        if(l <= n) return n - l + 1;
        return 0;
    }
    if(r < m) return r - l + 1;
    if(l < m) return m - l ;
    return 0;
}
ll query(int&id,int x,int l,int r)//id是当前在哪个节点  x是要求找到的当前树的哪个位置   l,r是当前节点的范围 
{
    if(!id)//如果还没建树 
    {
        id = ++SZ;//开一个新节点 
        sz[id] = get_sz(l,r)//建树 计算个数 
        if(l == r)
        {
            if(now <= n) val[id] = (ll)(now - 1) * m + l;//计算当前位置人的编号 
            else val[id] = (ll)l * m;
        }
    }
    sz[id]--;//因为当前是在查询，要删除一个人，所以个数-- 
    if(l == r) return val[id];//搜到最后了，返回当前人的编号 
    int mid = (l + r) >> 1;
    if((!ls[id] && x <= (mid - l + 1))/*因为如果左边还没去过即sz[ls[id]]为0，要特判*/ || x <= sz[ls[id]]) return query(ls[id],x,l,mid);//如果要查找的位置在左子树 
    else
    {
        if(!ls[id]) x -= (mid - l + 1); else x -= sz[ls[id]];//也是特判如果左边为0 
        return query(rs[id],x,mid + 1,r);
    }
}
void update(int&id,int x,ll w,int l,int r)//更新，即加入一个节点  人编号为w，x是要加入的位置 
{
    if(!id)//遍历到最后一个空位置，更新 
    {
        id = ++SZ;
        sz[id] = get_sz(l,r);
        if(l == r) val[id] = w;
    }
    sz[id]++;//加入一个节点，所以个数要++ 
    if(l == r) return ;//遍历到最后就返回 
    int mid = (l + r) >> 1;
    if(x <= mid) update(ls[id],x,w,l,mid);//朝着目标位置递归下去 
    else update(rs[id],x,w,mid + 1,r);
}
int main()
{
    init("phalanx");

    read(n); read(m); read(q);
    int p = max(m,n) + q ;ll z;

        if(y == m) now = n + 1,z = query(root[now],x,1,p);//查询 
        else now = x,z = query(root[now],y,1,p);
        printf("%lld\n",z);
        now = n + 1;update(root[n + 1],n + ( ++o[n+1] ),z,1,p); //把找出来的人插入最后一列的树的最后面 
        if(y != m)//如果找的位置不是最后一列，则还要把最后一列的第x个人插入到第x列的树的最后面 
        {
            now = n + 1;z = query(root[now],x,1,p);//先找到最后一列的第x个人 
            now = x;update(root[x],m - 1 + ( ++o[x] ),z,1,p);//插入到第x棵树里 
            //m - 1是原来的长度，++o[x]是后来新增的个数，两个加起来就是最后一位 ，上面同理 
        }
    }
    return 0;
}
